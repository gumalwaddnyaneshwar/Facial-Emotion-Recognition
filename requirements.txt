# DeepFER — requirements.txt
#
# These versions are pinned deliberately, not left open-ended — they
# reflect the exact combination that was proven to work together after
# resolving a real dependency conflict between TensorFlow, mediapipe,
# and jax (mediapipe pins protobuf<5 and older ml_dtypes; TensorFlow
# and Keras want newer versions of both). Installing "latest" for each
# package independently WILL reintroduce that conflict.
#
# Install with:
#   pip install -r requirements.txt
#
# You will likely see a pip dependency-resolver warning mentioning
# "tensorflow-intel requires keras<2.16" during install. This is
# expected and harmless — the app is confirmed working with the
# keras version pinned below despite that warning.

# ── Core ML pipeline ─────────────────────────────────────────────
tensorflow==2.15.1
keras==3.12.3
numpy>=1.23.5,<2.0
ml-dtypes~=0.3.1
protobuf>=4.25.3,<5
jax<0.4.26
jaxlib<0.4.26

# ── Computer vision / face detection ─────────────────────────────
mediapipe==0.10.14
# IMPORTANT: opencv-contrib-python-headless, NOT opencv-python or
# opencv-contrib-python. The regular (non-headless) builds expect
# graphical display libraries (libGL.so.1 etc.) that exist on a
# Windows desktop but NOT on Streamlit Cloud's headless Linux servers
# — using them there causes an ImportError at startup. The headless
# build is functionally identical for everything this app does (no
# GUI windows are used anywhere). Only ONE cv2 package should ever be
# listed here — opencv-python, opencv-contrib-python, and their
# headless variants all install into the same `cv2` module name, and
# having more than one installed risks a silent version conflict.
opencv-contrib-python-headless
Pillow

# ── App (Streamlit) ───────────────────────────────────────────────
streamlit
pandas

# ── PDF report export ─────────────────────────────────────────────
reportlab
matplotlib

# ── REST API (only needed if you also run/deploy api.py separately —
#    app.py, the file being deployed to Streamlit Cloud, does not
#    import these at all, so they add install time without being used
#    there. Harmless to leave in; remove if you want a faster Streamlit
#    Cloud build and aren't deploying the API too.) ──────────────────
fastapi
uvicorn
python-multipart
